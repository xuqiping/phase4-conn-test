### Python KANO分析脚本核心逻辑

```python
# KANO判定矩阵
KANO_MATRIX = {
    ('like','like'): 'Q', ('like','expect'): 'A', ('like','neutral'): 'A',
    ('like','tolerate'): 'A', ('like','dislike'): 'O',
    ('expect','like'): 'R', ('expect','expect'): 'I', ('expect','neutral'): 'I',
    ('expect','tolerate'): 'I', ('expect','dislike'): 'M',
    # ... 完整5x5矩阵
}

def kano_classify(responses):
    """输入正反向作答列表，输出KANO类别"""
    counts = {'M':0, 'O':0, 'A':0, 'I':0, 'R':0, 'Q':0}
    for positive, negative in responses:
        category = KANO_MATRIX[(positive, negative)]
        counts[category] += 1
    # Max频率法
    valid = {k:v for k,v in counts.items() if k != 'Q'}
    return max(valid, key=valid.get)

def better_worse(counts):
    """计算SI和DSI"""
    A, O, M, I = counts['A'], counts['O'], counts['M'], counts['I']
    total = A + O + M + I
    SI = (A + O) / total if total > 0 else 0
    DSI = -1 * (O + M) / total if total > 0 else 0
    return SI, DSI
```
