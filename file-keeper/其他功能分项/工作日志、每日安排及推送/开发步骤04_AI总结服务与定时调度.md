### 4.1 AI 总结服务

创建 `AiSummaryService`：

```java
@Service
@RequiredArgsConstructor
@Slf4j
public class AiSummaryService {

    private final RestTemplate restTemplate;

    @Value("${work-report.ai.provider:qwen}")
    private String aiProvider;

    @Value("${work-report.ai.api-key:}")
    private String apiKey;

    @Value("${work-report.ai.endpoint:}")
    private String endpoint;

    @Value("${work-report.ai.timeout-seconds:30}")
    private int timeoutSeconds;

    public String summarize(List<WorkLog> logs, List<WorkPlan> plans, String reportType) {
        if (logs.isEmpty() && plans.isEmpty()) {
            return "";
        }

        String prompt = buildPrompt(logs, plans, reportType);

        try {
            return callAiApi(prompt);
        } catch (Exception e) {
            log.error("AI 总结失败，降级为简单拼接", e);
            return fallbackSummary(logs, plans);
        }
    }

    private String buildPrompt(List<WorkLog> logs, List<WorkPlan> plans, String reportType) {
        StringBuilder sb = new StringBuilder();
        sb.append("请根据以下工作记录和计划，生成一份简洁规范的").append(reportType.equals("DAILY") ? "日报" : "周报").append("：\n\n");
        sb.append("【工作记录】\n");
        for (WorkLog log : logs) {
            sb.append("- ").append(log.getContent()).append("\n");
        }
        sb.append("\n【计划】\n");
        for (WorkPlan plan : plans) {
            sb.append("- ").append(plan.getContent())
              .append(plan.getCompleted() ? "（已完成）" : "（未完成）").append("\n");
        }
        sb.append("\n要求：\n");
        sb.append("1. 用第一人称\n");
        sb.append("2. 分").append(reportType.equals("DAILY") ? "今日工作、遇到的问题、明日计划" : "本周工作、遇到的问题、下周计划").append("三部分\n");
        sb.append("3. 语言简洁专业\n");
        return sb.toString();
    }

    private String callAiApi(String prompt) {
        // 根据 aiProvider 调用对应 API
        // 示例：通义千问
        return switch (aiProvider) {
            case "qwen" -> callQwen(prompt);
            case "doubao" -> callDoubao(prompt);
            case "claude" -> callClaude(prompt);
            default -> throw new BusinessException(ErrorCode.CONFIG_ERROR, "不支持的 AI 提供商");
        };
    }

    private String callQwen(String prompt) {
        // 实现 HTTP 调用
        return "AI 总结结果";
    }

    private String fallbackSummary(List<WorkLog> logs, List<WorkPlan> plans) {
        StringBuilder sb = new StringBuilder();
        sb.append("## 工作记录\n");
        logs.forEach(log -> sb.append("- ").append(log.getContent()).append("\n"));
        sb.append("\n## 计划\n");
        plans.forEach(plan -> sb.append("- ").append(plan.getContent()).append("\n"));
        return sb.toString();
    }
}
```

### 4.3 报告模板引擎

创建 `ReportTemplateEngine`：

```java
@Component
public class ReportTemplateEngine {

    public String render(String templateContent, Map<String, Object> context) {
        String result = templateContent;
        for (Map.Entry<String, Object> entry : context.entrySet()) {
            String placeholder = "{{" + entry.getKey() + "}}";
            String value = entry.getValue() != null ? entry.getValue().toString() : "";
            result = result.replace(placeholder, value);
        }
        return result;
    }

    public Map<String, Object> buildContext(
            String aiSummary,
            List<WorkLog> logs,
            List<WorkPlan> plans,
            String reportType) {
        Map<String, Object> context = new HashMap<>();
        context.put("ai_summary", aiSummary);
        context.put("logs", formatLogs(logs));
        context.put("plans", formatPlans(plans));
        context.put("report_type", reportType);
        context.put("generated_at", LocalDateTime.now().toString());
        return context;
    }

    private String formatLogs(List<WorkLog> logs) {
        return logs.stream()
            .map(log -> "- " + log.getContent())
            .collect(Collectors.joining("\n"));
    }

    private String formatPlans(List<WorkPlan> plans) {
        return plans.stream()
            .map(plan -> "- " + plan.getContent() + (plan.getCompleted() ? " ✓" : ""))
            .collect(Collectors.joining("\n"));
    }
}
```

### 4.4 报告生成服务

完善 `WorkReportService.generate()`：

```java
@Transactional
public WorkReportDto generate(Long userId, Long configId) {
    ReportConfig config = reportConfigService.getById(userId, configId);
    if (config == null) {
        throw new BusinessException(ErrorCode.NOT_FOUND, "报告配置不存在");
    }

    ReportTemplate template = reportTemplateService.getById(config.getTemplateId());
    if (template == null) {
        throw new BusinessException(ErrorCode.NOT_FOUND, "报告模板不存在");
    }

    // 确定日期范围
    DateRange range = calculateDateRange(config.getReportType());

    // 查询数据
    List<WorkLog> logs = workLogService.listByUserAndDateRange(userId, range.start(), range.end());
    List<WorkPlan> plans = workPlanService.listByUserAndDateRange(userId, range.start(), range.end());

    // AI 总结
    String aiSummary = config.getAiEnabled()
        ? aiSummaryService.summarize(logs, plans, config.getReportType())
        : "";

    // 渲染模板
    Map<String, Object> context = templateEngine.buildContext(aiSummary, logs, plans, config.getReportType());
    String content = templateEngine.render(template.getContent(), context);

    // 生成标题
    String title = generateTitle(config.getReportType(), range);

    // 保存报告
    WorkReport report = new WorkReport();
    report.setUserId(userId);
    report.setConfigId(configId);
    report.setReportType(config.getReportType());
    report.setTitle(title);
    report.setContent(content);
    report.setGeneratedAt(LocalDateTime.now());
    report.setStatus("GENERATED");
    workReportMapper.insert(report);

    return workReportMapper.toDto(report);
}
```

### 4.5 定时调度服务

创建 `ReportScheduleService`：

```java
@Service
@RequiredArgsConstructor
@Slf4j
public class ReportScheduleService {

    private final ReportConfigService reportConfigService;
    private final WorkReportService workReportService;
    private final PushService pushService;

    @Scheduled(cron = "0 * * * * ?")
    public void scheduleReports() {
        log.info("开始扫描报告配置...");
        List<ReportConfig> configs = reportConfigService.listEnabled();
        LocalDateTime now = LocalDateTime.now();

        for (ReportConfig config : configs) {
            try {
                if (shouldTrigger(config, now)) {
                    log.info("触发报告生成: configId={}", config.getId());
                    WorkReportDto report = workReportService.generate(config.getUserId(), config.getId());
                    pushService.pushReport(report.getId());
                }
            } catch (Exception e) {
                log.error("报告生成/推送失败: configId={}", config.getId(), e);
            }
        }
    }

    private boolean shouldTrigger(ReportConfig config, LocalDateTime now) {
        // 1. 校验 cron 表达式是否匹配当前时间（精确到分钟）
        // 2. 校验 timezone
        // 可使用 CronExpression 或 quartz 的 CronExpression
        return CronExpression.isValidExpression(config.getCronExpression())
            && new CronExpression(config.getCronExpression())
                .getNextValidTimeAfter(Date.from(now.atZone(ZoneId.of(config.getTimezone())).toInstant())) != null;
    }
}
```

> 注意：MVP 阶段可使用 Spring `@Scheduled` + `CronExpression`。如果后续需要分布式调度，再引入 Quartz 集群。

## 8. 风险与注意事项

| 风险 | 说明 |
|---|---|
| AI API 不稳定 | 必须加超时和降级，避免报告生成卡住 |
| Prompt 效果差 | MVP 先用简单 Prompt，后续根据用户反馈迭代 |
| 定时任务时区问题 | 统一使用 `Asia/Shanghai`，配置中保存 timezone |
| 报告生成重复触发 | 调度扫描精确到分钟，避免同一分钟重复触发 |
| 模板占位符遗漏 | 使用 `{{key}}` 约定，缺失时替换为空字符串 |
