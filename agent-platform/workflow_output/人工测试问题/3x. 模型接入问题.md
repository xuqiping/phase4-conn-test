```
# 1.上传素材
curl --location 'https://ai.ctaigw.cn/v1/services/aigc/video-generation/video-synthesis'  \
    -H 'X-DashScope-Async: enable' \
    -H "Authorization: Bearer $API_KEY" \
    -H 'Content-Type: application/json' \
    -d '{
    "model": "happyhorse-1.1-r2v",
    "input": {
        "prompt": "一只猫在草地上奔跑",
        "media": [
            {
                "type": "first_frame",
                "url": "https://cdn.translate.alibaba.com/r/wanx-demo-1.png"
            }
        ]
    },
    "parameters": {
        "resolution": "720P",
        "duration": 5
    }
}'

# 2. 获取视频任务结果
curl -X GET 'https://ai.ctaigw.cn/v1/tasks/{task_id}' -H 'Authorization: Bearer $API_KEY'
```
