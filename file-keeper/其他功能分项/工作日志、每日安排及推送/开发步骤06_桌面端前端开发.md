### 4.2 API 层

```ts
import { fetchR } from './commercialAuth'
import type { WorkLog, WorkPlan, ReportTemplate, ReportConfig, WorkReport } from '@/types/workReport'

const BASE_PATH = '/api/client/work-report'

export async function listWorkLogs(
  baseUrl: string,
  token: string,
  date: string
): Promise<WorkLog[]> {
  return fetchR<WorkLog[]>(`${baseUrl}${BASE_PATH}/logs?date=${date}`, token)
}

export async function createWorkLog(
  baseUrl: string,
  token: string,
  log: Partial<WorkLog>
): Promise<WorkLog> {
  return fetchR<WorkLog>(`${baseUrl}${BASE_PATH}/logs`, token, {
    method: 'POST',
    body: JSON.stringify(log),
  })
}

export async function updateWorkLog(
  baseUrl: string,
  token: string,
  id: number,
  log: Partial<WorkLog>
): Promise<WorkLog> {
  return fetchR<WorkLog>(`${baseUrl}${BASE_PATH}/logs/${id}`, token, {
    method: 'PUT',
    body: JSON.stringify(log),
  })
}

export async function deleteWorkLog(
  baseUrl: string,
  token: string,
  id: number
): Promise<void> {
  return fetchR<void>(`${baseUrl}${BASE_PATH}/logs/${id}`, token, {
    method: 'DELETE',
  })
}

export async function listWorkPlans(
  baseUrl: string,
  token: string,
  date: string
): Promise<WorkPlan[]> {
  return fetchR<WorkPlan[]>(`${baseUrl}${BASE_PATH}/plans?date=${date}`, token)
}

export async function saveWorkPlan(

  return fetchR<WorkPlan>(`${baseUrl}${BASE_PATH}/plans`, token, {

  return fetchR<ReportConfig>(`${baseUrl}${BASE_PATH}/configs`, token, {

  baseUrl: string,

  return fetchR<void>(`${baseUrl}${BASE_PATH}/configs/${id}`, token, {

  baseUrl: string,

  return fetchR<WorkReport>(`${baseUrl}${BASE_PATH}/reports/generate`, token, {
```

### 4.3 Store

新增 [`src/stores/workReportStore.ts`](../../src/stores/workReportStore.ts)：

```ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { useAuthStore } from './authStore'
import * as api from '@/api/workReport'
import type { WorkLog, WorkPlan, ReportTemplate, ReportConfig, WorkReport } from '@/types/workReport'

export const useWorkReportStore = defineStore('work-report', () => {
  const logs = ref<WorkLog[]>([])
  const plans = ref<WorkPlan[]>([])
  const templates = ref<ReportTemplate[]>([])
  const configs = ref<ReportConfig[]>([])
  const reports = ref<WorkReport[]>([])
  const currentReport = ref<WorkReport | null>(null)
  const currentDate = ref<string>(new Date().toISOString().split('T')[0])
  const loading = ref(false)
  const error = ref<string | null>(null)

  const todayLogs = computed(() => logs.value.sort((a, b) => (a.sortOrder ?? 0) - (b.sortOrder ?? 0)))
  const todayPlans = computed(() => plans.value.sort((a, b) => (a.sortOrder ?? 0) - (b.sortOrder ?? 0)))

  function getBaseUrlAndToken() {
    const authStore = useAuthStore()
    if (!authStore.baseUrl || !authStore.accessToken) {
      throw new Error('未登录')
    }
    return { baseUrl: authStore.baseUrl, token: authStore.accessToken }
  }

  async function loadToday(date?: string) {
    if (date) currentDate.value = date
    const { baseUrl, token } = getBaseUrlAndToken()
    loading.value = true
    error.value = null
    try {
      const [logsRes, plansRes] = await Promise.all([
        api.listWorkLogs(baseUrl, token, currentDate.value),
        api.listWorkPlans(baseUrl, token, currentDate.value),
      ])
      logs.value = logsRes
      plans.value = plansRes
    } catch (e) {
      error.value = e instanceof Error ? e.message : String(e)
    } finally {
      loading.value = false
    }
  }

  async function saveLog(log: Partial<WorkLog>) {
    const { baseUrl, token } = getBaseUrlAndToken()
    if (log.id) {
      await api.updateWorkLog(baseUrl, token, log.id, log)
    } else {
      await api.createWorkLog(baseUrl, token, { ...log, logDate: currentDate.value })
    }
    await loadToday()
  }

  async function removeLog(id: number) {
    const { baseUrl, token } = getBaseUrlAndToken()
    await api.deleteWorkLog(baseUrl, token, id)
    await loadToday()
  }

  async function savePlan(plan: Partial<WorkPlan>) {
    const { baseUrl, token } = getBaseUrlAndToken()
    await api.saveWorkPlan(baseUrl, token, { ...plan, planDate: currentDate.value })
    await loadToday()
  }

  async function removePlan(id: number) {
    const { baseUrl, token } = getBaseUrlAndToken()
    await api.deleteWorkPlan(baseUrl, token, id)
    await loadToday()
  }

  async function loadTemplates() {
    const { baseUrl, token } = getBaseUrlAndToken()
    templates.value = await api.listReportTemplates(baseUrl, token)
  }

  async function loadConfigs() {
    const { baseUrl, token } = getBaseUrlAndToken()
    configs.value = await api.listReportConfigs(baseUrl, token)
  }

  async function saveConfig(config: Partial<ReportConfig>) {
    const { baseUrl, token } = getBaseUrlAndToken()
    await api.saveReportConfig(baseUrl, token, config)
    await loadConfigs()
  }

  async function deleteConfig(id: number) {
    const { baseUrl, token } = getBaseUrlAndToken()
    await api.deleteReportConfig(baseUrl, token, id)
    await loadConfigs()
  }

  async function generateReport(configId: number) {
    const { baseUrl, token } = getBaseUrlAndToken()
    loading.value = true
    try {
      currentReport.value = await api.generateReport(baseUrl, token, configId)
      return currentReport.value
    } catch (e) {
      error.value = e instanceof Error ? e.message : String(e)
      throw e
    } finally {
      loading.value = false
    }
  }

  async function loadReports(page = 1, size = 20) {
    const { baseUrl, token } = getBaseUrlAndToken()
    const res = await api.listReports(baseUrl, token, page, size)
    reports.value = res.list
    return res
  }

  async function pushReport(reportId: number) {
    const { baseUrl, token } = getBaseUrlAndToken()
    await api.pushReport(baseUrl, token, reportId)
  }

  return {
    logs, plans, templates, configs, reports, currentReport,
    currentDate, loading, error,
    todayLogs, todayPlans,
    loadToday, saveLog, removeLog, savePlan, removePlan,
    loadTemplates, loadConfigs, saveConfig, deleteConfig,
    generateReport, loadReports, pushReport,
  }
})
```

## 8. 风险与注意事项

| 风险 | 说明 |
|---|---|
| Tab 过多影响布局 | 当前已有 files/processes/clipboard，增加 work-report 后需确认底部/顶部 Tab 空间 |
| 日期选择器国际化 | 使用 Naive UI 的 `n-date-picker`，注意与现有主题适配 |
| cron 表达式用户难理解 | UI 上提供预设选项，高级用户才手动输入 |
| 报告内容 Markdown 渲染 | 可引入轻量 Markdown 渲染库，或先用 `<pre>` 展示 |
| Store 与后端状态同步 | 每次操作后重新加载当日数据，避免状态不一致 |
